cmake_minimum_required(VERSION 3.16)
project(DuskStreet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(glfw3 3.3 REQUIRED)
add_executable(dusk-street src/main.cpp src/Scene.cpp src/Mesh.cpp
    src/Tour.cpp src/Camera.cpp src/ShaderProgram.cpp src/Texture2D.cpp)
target_include_directories(dusk-street PRIVATE src)
target_include_directories(dusk-street SYSTEM PRIVATE third_party/glm third_party/stb)
target_link_libraries(dusk-street PRIVATE glfw GLEW::GLEW OpenGL::GL)
target_compile_definitions(dusk-street PRIVATE STREET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(dusk-street PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
enable_testing()
add_test(NAME camera-tour COMMAND dusk-street --test-tour)
